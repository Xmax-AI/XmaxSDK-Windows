cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" XMAX_PACKAGE_VERSION LIMIT_COUNT 1)
project(XmaxWindows VERSION ${XMAX_PACKAGE_VERSION} LANGUAGES C CXX)
if(MSVC)
  # Keep UTF-8 literals and raw-string macro arguments independent of the Windows locale.
  add_compile_options(/utf-8 /Zc:preprocessor)
endif()
if(NOT WIN32 AND NOT APPLE)
  message(FATAL_ERROR "Supported build hosts are Windows and macOS.")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "XmaxSDK currently supports Windows x64 only.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
include(CTest)
option(XMAX_BUILD_EXAMPLE "Build Qt 6 X-Lab desktop example" OFF)
option(XMAX_WITH_AGORA "Enable the native Agora 4.x adapter" OFF)
option(XMAX_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(XMAX_AGORA_ROOT "" CACHE PATH "Extracted Agora SDK directory containing high_level_api/include and x86_64")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(XMAX_ENABLE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(Sources/XmaxSDK)
if(BUILD_TESTING)
  # This independent C executable is compared with managed sizeof/offsetof in verify-dotnet.ps1.
  add_executable(xmax_dotnet_abi_probe Tests/XmaxSDKNetTests/native_abi_probe.c)
  target_include_directories(xmax_dotnet_abi_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
endif()
if(XMAX_BUILD_EXAMPLE)
  add_subdirectory(Examples/XLab)
  set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT XLab)
endif()
