cmake_minimum_required(VERSION 3.24)
project(XmaxWindows VERSION 0.2.0 LANGUAGES C CXX)
include(CTest)
option(XMAX_BUILD_EXAMPLE "Build Qt 6 X-Lab desktop example" OFF)
option(XMAX_WITH_AGORA "Enable the native Agora 4.x adapter" OFF)
option(XMAX_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(XMAX_AGORA_ROOT "" CACHE PATH "Extracted Agora SDK directory containing high_level_api/include and x86_64")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(XMAX_ENABLE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(Sources/XmaxSDK)
if(XMAX_BUILD_EXAMPLE)
  add_subdirectory(Examples/XLab)
endif()
