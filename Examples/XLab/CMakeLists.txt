find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Multimedia Concurrent Network)
if(APPLE AND TARGET WrapOpenGL::WrapOpenGL)
  # Qt 6.8 advertises AGL although modern SDKs no longer ship its link stub.
  # X-Lab uses Qt Quick's renderer and has no AGL symbol references.
  get_target_property(XLAB_OPENGL_LIBS WrapOpenGL::WrapOpenGL INTERFACE_LINK_LIBRARIES)
  list(FILTER XLAB_OPENGL_LIBS EXCLUDE REGEX "AGL")
  set_property(TARGET WrapOpenGL::WrapOpenGL PROPERTY INTERFACE_LINK_LIBRARIES "${XLAB_OPENGL_LIBS}")
endif()
qt_standard_project_setup(REQUIRES 6.8)
add_library(xlab_recording STATIC recording.cpp recording.hpp recording_writer.cpp)
target_link_libraries(xlab_recording PUBLIC Xmax::SDK Qt6::Multimedia Qt6::Gui)
if(WIN32)
  target_link_libraries(xlab_recording PRIVATE mfreadwrite mfplat mfuuid ole32)
endif()
if(MSVC)
  target_compile_options(xlab_recording PRIVATE /utf-8 /W4)
endif()
qt_add_executable(XLab main.cpp controller.cpp controller.hpp rounded_image.cpp rounded_image.hpp)
qt_add_qml_module(XLab URI XLab VERSION 1.0
  QML_FILES qml/Main.qml qml/Pill.qml qml/Icon.qml qml/CameraPicker.qml qml/Checkerboard.qml qml/VideoPanel.qml qml/ReferenceList.qml qml/XLabToolTip.qml qml/PageEntry.qml
  RESOURCES assets/presets.json assets/xmax-wordmark.png assets/DMSans.ttf assets/DMSans-OFL.txt)
target_link_libraries(XLab PRIVATE xlab_recording Xmax::SDK Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Concurrent Qt6::Network)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  target_link_libraries(XLab PRIVATE Qt6::Test)
  target_compile_definitions(XLab PRIVATE XLAB_TEST_SUPPORT=1)
  add_test(NAME xlab_ui COMMAND XLab --smoke-test --screenshot-dir "${CMAKE_BINARY_DIR}/screenshots")
  add_test(NAME xlab_key_gate COMMAND XLab --smoke-test-key-gate --screenshot-dir "${CMAKE_BINARY_DIR}/screenshots")
  qt_add_executable(xlab_behavior_tests tests/behavior_tests.cpp controller.cpp controller.hpp rounded_image.cpp rounded_image.hpp)
  target_compile_definitions(xlab_behavior_tests PRIVATE XLAB_TEST_SUPPORT=1)
  qt_add_resources(xlab_behavior_tests "test_presets" PREFIX "/qt/qml/XLab" FILES assets/presets.json)
  target_include_directories(xlab_behavior_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(xlab_behavior_tests PRIVATE xlab_recording Xmax::SDK Qt6::Quick Qt6::Multimedia Qt6::Concurrent Qt6::Network Qt6::Test)
  if(MSVC)
    target_compile_options(xlab_behavior_tests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME xlab_behavior COMMAND xlab_behavior_tests)
  qt_add_executable(xlab_recording_tests tests/recording_tests.cpp controller.cpp controller.hpp)
  target_compile_definitions(xlab_recording_tests PRIVATE XLAB_TEST_SUPPORT=1)
  target_include_directories(xlab_recording_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(xlab_recording_tests PRIVATE xlab_recording Qt6::Quick Qt6::Concurrent Qt6::Network Qt6::Test)
  add_test(NAME xlab_recording COMMAND xlab_recording_tests)
  set_tests_properties(xlab_ui xlab_key_gate xlab_behavior xlab_recording PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 60)
  if(WIN32 AND DEFINED ENV{SystemRoot})
    # Offscreen does not discover the native Windows font database automatically.
    file(TO_CMAKE_PATH "$ENV{SystemRoot}/Fonts" XLAB_SYSTEM_FONTS)
    set_property(TEST xlab_ui xlab_key_gate xlab_behavior xlab_recording APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=${XLAB_SYSTEM_FONTS}")
  endif()
endif()
