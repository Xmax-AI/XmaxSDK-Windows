function(xmax_module name)
  add_library(${name} STATIC ${ARGN})
  target_include_directories(${name} PUBLIC "${XMAX_REPO_ROOT}/include" PRIVATE "${XMAX_SRC}" "${XMAX_REPO_ROOT}/third_party")
  set_target_properties(${name} PROPERTIES CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  if(WIN32)
    target_compile_definitions(${name} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  endif()
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /utf-8 "$<$<CONFIG:Release>:/Z7>")
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

if(APPLE)
  # Older libc++ headers can force default visibility on template instantiations.
  # Limit the shared-library boundary to the public C entry points on every toolchain.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${XMAX_REPO_ROOT}/include/xmax/xmax.h")
  file(READ "${XMAX_REPO_ROOT}/include/xmax/xmax.h" XMAX_PUBLIC_HEADER)
  string(REGEX MATCHALL "XMAX_CALL[ \t\r\n]+xmax_[A-Za-z0-9_]+[ \t\r\n]*\\(" XMAX_PUBLIC_FUNCTIONS "${XMAX_PUBLIC_HEADER}")
  set(XMAX_EXPORTED_SYMBOLS "")
  foreach(function IN LISTS XMAX_PUBLIC_FUNCTIONS)
    string(REGEX REPLACE "XMAX_CALL[ \t\r\n]+(xmax_[A-Za-z0-9_]+)[ \t\r\n]*\\(" "_\\1\n" symbol "${function}")
    string(APPEND XMAX_EXPORTED_SYMBOLS "${symbol}")
  endforeach()
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/XmaxSDK.exports" CONTENT "${XMAX_EXPORTED_SYMBOLS}")
  target_link_options(XmaxSDK PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_BINARY_DIR}/XmaxSDK.exports")
  set_property(TARGET XmaxSDK APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/XmaxSDK.exports")
endif()
