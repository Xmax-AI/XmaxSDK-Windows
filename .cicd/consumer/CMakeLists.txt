cmake_minimum_required(VERSION 3.24)
project(XmaxInstalledConsumer LANGUAGES C CXX)
find_package(XmaxSDK CONFIG REQUIRED)
set(XMAX_CONSUMER_SOURCES "${CMAKE_CURRENT_LIST_DIR}")
if(NOT EXISTS "${XMAX_CONSUMER_SOURCES}/c_consumer.c")
  set(XMAX_CONSUMER_SOURCES "${CMAKE_CURRENT_LIST_DIR}/../../Tests/XmaxSDKTests")
endif()
add_executable(c_host "${XMAX_CONSUMER_SOURCES}/c_consumer.c")
target_link_libraries(c_host PRIVATE Xmax::SDK)
add_executable(cpp_host "${XMAX_CONSUMER_SOURCES}/cpp_consumer.cpp")
target_compile_features(cpp_host PRIVATE cxx_std_17)
target_link_libraries(cpp_host PRIVATE Xmax::SDK)
enable_testing()
add_test(NAME installed_c COMMAND c_host)
add_test(NAME installed_cpp COMMAND cpp_host)
if(WIN32)
  # Import libs do not put the DLL on PATH. Test an independently installed SDK.
  get_target_property(XMAX_INCLUDE_DIRS Xmax::SDK INTERFACE_INCLUDE_DIRECTORIES)
  list(GET XMAX_INCLUDE_DIRS 0 XMAX_INCLUDE_DIR)
  get_filename_component(XMAX_PREFIX "${XMAX_INCLUDE_DIR}/.." ABSOLUTE)
  foreach(target IN ITEMS c_host cpp_host)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${XMAX_PREFIX}/bin" "$<TARGET_FILE_DIR:${target}>")
  endforeach()
endif()
